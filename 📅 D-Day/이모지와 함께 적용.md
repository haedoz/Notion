### 이모지+색상 style
><img width="491" height="235" alt="스크린샷 2026-07-27 오전 11 37 30" src="https://github.com/user-attachments/assets/68de4477-aedc-43ae-996a-f36c704a885a" />

```javascript
lets(
  d, dateBetween(prop("마감일"), today(), "days"),
  
  if(
    d == 0,
    style("🔥 D-Day","b","green"),
    
    if(
      d <= 3 and d > 0,
      style("⚠️ D-" + format(d),"b","red"),
      
      if(
        d > 0,
        style("📅 D-" + format(d),"blue"),
        
        style("✅ D+" + format(abs(d)),"gray")
      )
    )
  )
)
```
