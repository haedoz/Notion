### 👾 소스코드 : 시간/분
><img width="561" height="125" alt="스크린샷 2026-08-07 오후 5 59 52" src="https://github.com/user-attachments/assets/8db3ed92-a7b5-4037-9cfb-776f8e4d1031" />

```javascript
lets(
  start, replaceAll(시작,":",""),
  end, replaceAll(종료,":",""),
  if(
    and(empty(start), empty(end)),
    "시작·종료 시간을 입력하세요".style("gray"),
    if(
      empty(start),
      "시작 시간을 입력하세요".style("gray"),
      if(
        empty(end),
        "종료 시간을 입력하세요".style("gray"),
        lets(
          startNum, toNumber(start),
          endNum, toNumber(end),
          startHour, floor(startNum / 100),
          startMinute, mod(startNum,100),
          endHour, floor(endNum / 100),
          endMinute, mod(endNum,100),
          startTotal, startHour * 60 + startMinute,
          endTotal, endHour * 60 + endMinute,
          diff, if(
            endTotal >= startTotal,
            endTotal - startTotal,
            endTotal + 1440 - startTotal
          ),
          if(
            diff < 60,
            format(diff) + "분",
            if(
              mod(diff,60) == 0,
              format(floor(diff/60)) + "시간",
              format(floor(diff/60)) + "시간 " + format(mod(diff,60)) + "분"
            )
          )
        )
      )
    )
  )
)
```
